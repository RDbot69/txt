EXPERIMENT 4: 4-BIT BINARY TO GRAY CODE CONVERTER
============================================================

DESIGN CODE (binary_to_gray.v)

module binary_to_gray (
    input  wire [3:0] binary,
    output wire [3:0] gray
);
    assign gray[3] = binary[3];
    assign gray[2] = binary[3] ^ binary[2];
    assign gray[1] = binary[2] ^ binary[1];
    assign gray[0] = binary[1] ^ binary[0];
endmodule

TESTBENCH CODE (tb_binary_to_gray.v)

`timescale 1ns / 1ps
module tb_binary_to_gray;
    reg  [3:0] binary;
    wire [3:0] gray;
    integer i;

    binary_to_gray DUT (.binary(binary), .gray(gray));

    initial begin
        $monitor("time=%0t binary=%b gray=%b", $time, binary, gray);
        for (i = 0; i < 16; i = i + 1) begin
            binary = i;
            #10;
        end
        $finish;
    end
endmodule

EXPERIMENT 4 - ADDITIONAL QUESTION: BCD TO COMMON-ANODE 7-SEGMENT DECODER
------------------------------------------------------------

Segment order in this code is {a,b,c,d,e,f,g}. Common-anode outputs are active-low: 0 turns a segment ON and 1 turns it OFF. The patterns below match the uploaded write-up.

DESIGN CODE (bcd_to_7segment.v)

module bcd_to_7segment (
    input  wire [3:0] bcd,
    output reg  [6:0] seg
);
    always @(*) begin
        case (bcd)
            4'b0000: seg = 7'b0000001; // 0
            4'b0001: seg = 7'b1001111; // 1
            4'b0010: seg = 7'b0010010; // 2
            4'b0011: seg = 7'b0000110; // 3
            4'b0100: seg = 7'b1001100; // 4
            4'b0101: seg = 7'b0100100; // 5
            4'b0110: seg = 7'b0100000; // 6
            4'b0111: seg = 7'b0001111; // 7
            4'b1000: seg = 7'b0000000; // 8
            4'b1001: seg = 7'b0000100; // 9
            default: seg = 7'b1111111; // Blank
        endcase
    end
endmodule

TESTBENCH CODE (tb_bcd_to_7segment.v)

`timescale 1ns / 1ps
module tb_bcd_to_7segment;
    reg  [3:0] bcd;
    wire [6:0] seg;
    integer i;

    bcd_to_7segment DUT (.bcd(bcd), .seg(seg));

    initial begin
        $monitor("time=%0t bcd=%d seg=%b", $time, bcd, seg);
        for (i = 0; i <= 9; i = i + 1) begin
            bcd = i;
            #10;
        end
        $finish;
    end
endmodule
