// ==================== DESIGN MODULE ====================
module ALU_4bit (
    input     [3:0] A,
    input     [3:0] B,
    input     [2:0] sel,
    output reg [3:0] Y
);

always @(*)
begin
    if (sel == 3'b000)             // Addition
        Y = A + B;
    else if (sel == 3'b001)        // Subtraction
        Y = A - B;
    else if (sel == 3'b010)        // AND
        Y = A & B;
    else if (sel == 3'b011)        // OR
        Y = A | B;
    else if (sel == 3'b100)        // XOR
        Y = A ^ B;
    else if (sel == 3'b101)        // NOT
        Y = ~A;
    else if (sel == 3'b110)        // Left Shift
        Y = A << 1;
    else if (sel == 3'b111)        // Right Shift
        Y = A >> 1;
    else
        Y = 4'b0000;
end

endmodule

// ==================== ORIGINAL TESTBENCH ====================
`timescale 1ns / 1ps

module ALU_4bit_tb;

reg [3:0] A;
reg [3:0] B;
reg [2:0] sel;
wire [3:0] Y;

// Instantiate the ALU
ALU_4bit uut (
    .A(A),
    .B(B),
    .sel(sel),
    .Y(Y)
);

initial
begin
    // Addition
    A = 4'b0101;
    B = 4'b0011;
    sel = 3'b000;
    #10;

    // Subtraction
    sel = 3'b001;
    #10;

    // AND
    sel = 3'b010;
    #10;

    // OR
    sel = 3'b011;
    #10;

    // XOR
    sel = 3'b100;
    #10;

    // NOT A
    sel = 3'b101;
    #10;

    // Left Shift
    sel = 3'b110;
    #10;

    // Right Shift
    sel = 3'b111;
    #10;

    // End simulation
    $stop;
end

endmodule

// Original source: alu_with carry.pdf — Experiment-03: ALU WITH CARRY
// Transcribed from the attached write-up. Page-wrap whitespace normalized.

// ==================== DESIGN MODULE ====================
module alu_4bit (
    input [3:0] A,
    input [3:0] B,
    input [2:0] sel,
    output reg [3:0] Y,
    output reg carry
);

reg [4:0] temp;

always @(*) begin
    Y = 4'b0000;
    carry = 1'b0;
    temp = 5'b00000;

    case (sel)
        3'b000: begin
            temp = A + B;
            Y = temp[3:0];
            carry = temp[4];
        end

        3'b001: begin
            Y = A - B;
            carry = 1'b0;
        end

        3'b010: Y = A & B;
        3'b011: Y = A | B;
        3'b100: Y = A ^ B;
        3'b101: Y = ~A;
        3'b110: Y = A << 1;
        3'b111: Y = A >> 1;
    endcase
end

endmodule

// ==================== ORIGINAL TESTBENCH ====================
`timescale 1ns/1ps

module alu_4bit_tb;

reg [3:0] A, B;
reg [2:0] sel;
wire [3:0] Y;
wire carry;

alu_4bit uut (
    .A(A),
    .B(B),
    .sel(sel),
    .Y(Y),
    .carry(carry)
);

initial begin
    A = 4'b1111;
    B = 4'b0001;

    sel = 3'b000; #10;
    sel = 3'b001; #10;
    sel = 3'b010; #10;
    sel = 3'b011; #10;
    sel = 3'b100; #10;
    sel = 3'b101; #10;
    sel = 3'b110; #10;
    sel = 3'b111; #10;
    $finish;
end

endmodule
