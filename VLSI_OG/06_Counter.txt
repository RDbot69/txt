PRACTICAL 6 - 3-BIT SYNCHRONOUS UP/DOWN COUNTER
===============================================

SOURCE FILE: 01_3_bit_Synchronous_Up_Down_Counter_with_Testbench_ORIGINAL.txt
----------------------------------------------------------------------
module up_down_counter_3bit (
     input wire clk,
     input wire rst_n,
     input wire up_down,
     output reg [2:0] count,
     output wire clk_div
);

     reg [25:0] counter;
     reg clk_div_reg;

     always @(posedge clk or negedge rst_n) begin
       if (!rst_n) begin
           counter <= 26'd0;
           clk_div_reg <= 1'b0;
       end
       else if (counter == 26'd49_999_999) begin
           counter <= 26'd0;
           clk_div_reg <= ~clk_div_reg;
       end
       else begin
           counter <= counter + 1'b1;
       end
     end

     assign clk_div = clk_div_reg;

     always @(posedge clk_div or negedge rst_n) begin
       if (!rst_n) begin
           count <= 3'b000;
       end
       else begin
           if (up_down)
             count <= count + 1'b1;
           else
             count <= count - 1'b1;
       end
     end

endmodule

`timescale 1ns / 1ps

module tb_up_down_counter_3bit;

  reg clk;
  reg rst_n;
  reg up_down;

  wire [2:0] count;
  wire clk_div;

  up_down_counter_3bit uut (
    .clk(clk),
    .rst_n(rst_n),
    .up_down(up_down),
    .count(count),
    .clk_div(clk_div)
  );

  always #5 clk = ~clk;

  initial begin
       clk = 0;
       rst_n = 0;
       up_down = 1;

       #20;
       rst_n = 1;

       @(posedge clk_div);
       @(posedge clk_div);
       @(posedge clk_div);
       @(posedge clk_div);

       up_down = 0;

       @(posedge clk_div);
       @(posedge clk_div);
       @(posedge clk_div);
       @(posedge clk_div);

       #100;
       $finish;
  end

endmodule
