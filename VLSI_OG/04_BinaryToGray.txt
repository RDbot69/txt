// ==================== DESIGN MODULE ====================
module binary_to_gray(
    input [3:0] binary,
    output [3:0] gray
);

assign gray[3] = binary[3];
assign gray[2] = binary[3] ^ binary[2];
assign gray[1] = binary[2] ^ binary[1];
assign gray[0] = binary[1] ^ binary[0];

endmodule

// ==================== ORIGINAL TESTBENCH ====================
module tb_binary_to_gray;

reg [3:0] binary;
wire [3:0] gray;

binary_to_gray DUT (
    .binary(binary),
    .gray(gray)
);

initial begin
    binary = 4'b0000; #10;
    binary = 4'b0001; #10;
    binary = 4'b0010; #10;
    binary = 4'b0011; #10;
    binary = 4'b0100; #10;
    binary = 4'b0101; #10;
    binary = 4'b0110; #10;
    binary = 4'b0111; #10;
    binary = 4'b1000; #10;
    binary = 4'b1001; #10;
    binary = 4'b1010; #10;
    binary = 4'b1011; #10;
    binary = 4'b1100; #10;
    binary = 4'b1101; #10;
    binary = 4'b1110; #10;
    binary = 4'b1111; #10;

    $finish;
end

endmodule

// ==================== DESIGN MODULE ====================
module bcd_to_7segment(
    input [3:0] bcd,
    output reg [6:0] seg
);

always @(bcd)
begin
    case(bcd)
        4'b0000: seg = 7'b0000001; // 0
        4'b0001: seg = 7'b1001111; // 1
        4'b0010: seg = 7'b0010010; // 2
        4'b0011: seg = 7'b0000110; // 3
        4'b0100: seg = 7'b1001100; // 4
        4'b0101: seg = 7'b0100100; // 5
        4'b0110: seg = 7'b0100000; // 6
        4'b0111: seg = 7'b0001111; // 7
        4'b1000: seg = 7'b0000000; // 8
        4'b1001: seg = 7'b0000100; // 9
        default: seg = 7'b1111111;
    endcase
end

endmodule

// ==================== ORIGINAL TESTBENCH ====================
module tb_bcd_to_7segment;

reg [3:0] bcd;
wire [6:0] seg;

bcd_to_7segment DUT (
    .bcd(bcd),
    .seg(seg)
);

initial
begin
    bcd = 4'b0000; #10;
    bcd = 4'b0001; #10;
    bcd = 4'b0010; #10;
    bcd = 4'b0011; #10;
    bcd = 4'b0100; #10;
    bcd = 4'b0101; #10;
    bcd = 4'b0110; #10;
    bcd = 4'b0111; #10;
    bcd = 4'b1000; #10;
    bcd = 4'b1001; #10;

    $finish;
end

endmodule
